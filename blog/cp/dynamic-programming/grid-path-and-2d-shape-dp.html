    Grid Path DP & Two-Dimensional Shape DP

    October 2026 · 25 min read · Patterns 9–10 (8 Problems)

    Connection #6 — A grid is a data structure, not a DP pattern: When a problem gives you a 2D matrix, do not assume a single formula. Ask what the state represents:
    • Grid Path DP (Pattern 9): Optimizes or counts routes moving along directed grid edges.
    • Maximal Square (Pattern 10): Uses 3-neighbor local bottleneck DP ($\min$ of top, left, top-left).
    • Maximal Rectangle (Pattern 10): Accumulates row-wise histogram heights and applies a monotonic stack.

    Pattern 9: Grid Path DP — Minimum or Maximum Cost

    How to Recognize It

    You move through a 2D grid or triangular structure under directional movement rules (e.g., right/down or falling down-left/down/down-right) and optimize path cost, survival health, or total path count.

    Core Template

    $$\text{dp}[r][c] = \text{grid}[r][c] + \min\big(\text{dp}[r - 1][c],\; \text{dp}[r][c - 1]\big)$$

    ▶ Step-Through: Backward Health DP in Dungeon Game (LC 174)

    Why forward DP fails on Dungeon Game: maximizing sum so far vs. maximizing minimum prefix dip do not have optimal substructure forward! Instead, work backward from the bottom-right princess cell: need[r][c] = max(1, min(need[r+1][c], need[r][c+1]) - dungeon[r][c]).

    9.1 62. Unique Paths

    State: $\text{dp}[r][c] = \text{dp}[r-1][c] + \text{dp}[r][c-1]$. In 1D rolling space, `dp[c] += dp[c - 1]`.

    class Solution {
public:
    int uniquePaths(int m, int n) {
        vector<int> dp(n, 1);
        for (int r = 1; r < m; ++r)
            for (int c = 1; c < n; ++c)
                dp[c] += dp[c - 1];
        return dp[n - 1];
    }
}; // Time: O(m * n), Space: O(n)

    9.2 63. Unique Paths II

    Blocked Cells: If `obstacleGrid[r][c] == 1`, immediately set `dp[c] = 0`; otherwise accumulate `dp[c] += dp[c - 1]`.

    class Solution {
public:
    int uniquePathsWithObstacles(vector<vector<int>>& grid) {
        int m = grid.size(), n = grid[0].size();
        vector<long long> dp(n, 0);
        dp[0] = (grid[0][0] == 0);
        for (int r = 0; r < m; ++r) {
            for (int c = 0; c < n; ++c) {
                if (grid[r][c] == 1) dp[c] = 0;
                else if (c > 0) dp[c] += dp[c - 1];
            }
        }
        return dp[n - 1];
    }
}; // Time: O(m * n), Space: O(n)

    9.3 64. Minimum Path Sum

    State: $\text{dp}[r][c] = \text{grid}[r][c] + \min(\text{dp}[r-1][c], \text{dp}[r][c-1])$.

    class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int m = grid.size(), n = grid[0].size();
        vector<int> dp(n, INT_MAX);
        dp[0] = 0;
        for (int r = 0; r < m; ++r) {
            dp[0] += grid[r][0];
            for (int c = 1; c < n; ++c)
                dp[c] = grid[r][c] + min(dp[c], dp[c - 1]);
        }
        return dp[n - 1];
    }
}; // Time: O(m * n), Space: O(n)

    9.4 120. Triangle

    Bottom-Up Triangle Collapse: Starting from the bottom row and folding upward via $\text{dp}[c] = \text{tri}[r][c] + \min(\text{dp}[c], \text{dp}[c+1])$ avoids boundary checks and lands the answer cleanly in `dp[0]`.

    class Solution {
public:
    int minimumTotal(vector<vector<int>>& triangle) {
        vector<int> dp = triangle.back();
        for (int r = (int)triangle.size() - 2; r >= 0; --r) {
            for (int c = 0; c <= r; ++c)
                dp[c] = triangle[r][c] + min(dp[c], dp[c + 1]);
        }
        return dp[0];
    }
}; // Time: O(n^2), Space: O(n)

    9.5 931. Minimum Falling Path Sum

    3-Way Predecessor Transition: Cell $(r, c)$ can be reached from $(r-1, c-1)$, $(r-1, c)$, or $(r-1, c+1)$.

    class Solution {
public:
    int minFallingPathSum(vector<vector<int>>& matrix) {
        int n = matrix.size();
        vector<int> dp = matrix[0];
        for (int r = 1; r < n; ++r) {
            vector<int> next_dp(n);
            for (int c = 0; c < n; ++c) {
                int best = dp[c];
                if (c > 0) best = min(best, dp[c - 1]);
                if (c + 1 < n) best = min(best, dp[c + 1]);
                next_dp[c] = matrix[r][c] + best;
            }
            dp = move(next_dp);
        }
        return *min_element(dp.begin(), dp.end());
    }
}; // Time: O(n^2), Space: O(n)

    9.6 174. Dungeon Game

    Key Observation — Backward DP from Destination: Forward DP cannot know whether a high current health with a worse future path is better than a lower current health with a gentle suffix. Working backward from $(m-1, n-1)$ solves this cleanly: $\text{dp}[r][c]$ = minimum health required upon entering $(r, c)$ to survive to the end:

    $$\text{dp}[r][c] = \max\Big(1,\; \min(\text{dp}[r + 1][c],\; \text{dp}[r][c + 1]) - \text{dungeon}[r][c]\Big)$$ 
    class Solution {
public:
    int calculateMinimumHP(vector<vector<int>>& dungeon) {
        int m = dungeon.size(), n = dungeon[0].size();
        vector<vector<int>> dp(m + 1, vector<int>(n + 1, INT_MAX));
        dp[m][n - 1] = dp[m - 1][n] = 1;
        for (int r = m - 1; r >= 0; --r) {
            for (int c = n - 1; c >= 0; --c) {
                int need = min(dp[r + 1][c], dp[r][c + 1]) - dungeon[r][c];
                dp[r][c] = max(1, need);
            }
        }
        return dp[0][0];
    }
}; // Time: O(m * n), Space: O(m * n) (easily O(n))

    Pattern 10: Two-Dimensional Shape DP

    How to Recognize It

    The problem asks you to find the largest valid square or rectangle of `'1'`s inside a binary matrix.

    10.1 221. Maximal Square

    3-Neighbor Bottleneck Recurrence: Let $\text{dp}[r][c]$ be the side length of the largest all-`1` square whose bottom-right corner is $(r, c)$. If `matrix[r][c] == '1'`:

    $$\text{dp}[r][c] = 1 + \min\big(\text{dp}[r - 1][c],\; \text{dp}[r][c - 1],\; \text{dp}[r - 1][c - 1]\big)$$ 
    class Solution {
public:
    int maximalSquare(vector<vector<char>>& matrix) {
        int m = matrix.size(), n = matrix[0].size(), maxSide = 0;
        vector<int> dp(n + 1, 0);
        for (int r = 1; r <= m; ++r) {
            int diag = 0;
            for (int c = 1; c <= n; ++c) {
                int tmp = dp[c];
                if (matrix[r - 1][c - 1] == '1') {
                    dp[c] = 1 + min({dp[c], dp[c - 1], diag});
                    maxSide = max(maxSide, dp[c]);
                } else {
                    dp[c] = 0;
                }
                diag = tmp;
            }
        }
        return maxSide * maxSide;
    }
}; // Time: O(m * n), Space: O(n)

    10.2 85. Maximal Rectangle

    Row-Wise Histogram DP + Monotonic Stack: Maintain $\text{height}[c]$ = consecutive `'1'`s ending at row $r$ in column $c$ ($\text{height}[c] = \text{matrix}[r][c] == \text{'1'} ? \text{height}[c] + 1 : 0$), then solve Largest Rectangle in Histogram on each row in $O(n)$ using a monotonic increasing stack.

    class Solution {
public:
    int maximalRectangle(vector<vector<char>>& matrix) {
        if (matrix.empty()) return 0;
        int m = matrix.size(), n = matrix[0].size(), maxArea = 0;
        vector<int> h(n + 1, 0); // trailing 0 sentinel flushes stack
        for (int r = 0; r < m; ++r) {
            for (int c = 0; c < n; ++c)
                h[c] = (matrix[r][c] == '1') ? h[c] + 1 : 0;
            vector<int> st;
            for (int c = 0; c <= n; ++c) {
                while (!st.empty() && h[st.back()] >= h[c]) {
                    int height = h[st.back()];
                    st.pop_back();
                    int width = st.empty() ? c : (c - st.back() - 1);
                    maxArea = max(maxArea, height * width);
                }
                st.push_back(c);
            }
        }
        return maxArea;
    }
}; // Time: O(m * n), Space: O(n)
