In Pattern 11 (Palindrome DP), an interval $[i, j]$ always transitioned deterministically to the single inner interval $[i+1, j-1]$ in $O(1)$ time. In Interval DP (Pattern 15) and Partition DP (Pattern 16), we do not know where the boundary lies, so we must search over all intermediate split points $k$, yielding $O(n^3)$ or $O(K \cdot n^2)$ recurrences. (See also our dedicated DSA Partition & Interval DP notes for additional matrix chain multiplication examples.)

1. Pattern 15: Interval DP — Choose a Split Point

In all three problems of Pattern 15, we define dp[i][j] over an open interval of interior choices bounded by fixed endpoints $i$ and $j$ (with $j - i \ge 2$). To solve $(i, j)$, we try every interior pivot $k \in (i+1, j-1)$, combine the independent left and right subproblems dp[i][k] and dp[k][j], and add the interaction cost determined by the triple $(i, k, j)$: 

// Core Recurrence Shape (Open-Interval Split DP):
for (int len = 2; len < m; ++len) {           // increasing interval length j - i
    for (int i = 0; i + len < m; ++i) {
        int j = i + len;
        for (int k = i + 1; k < j; ++k) {     // try every interior split point k
            dp[i][j] = best(dp[i][j], dp[i][k] + dp[k][j] + cost(i, k, j));
        }
    }
}
Why Open Intervals $(i, j)$ with Sentinel Endpoints?

Notice that in dp[i][k] + dp[k][j] + cost(i, k, j), the boundary indices $i$ and $j$ remain untouched until the very end of the subproblem, and the split point $k$ becomes the shared boundary of both child intervals $(i, k)$ and $(k, j)$:

  • 1039. Polygon Triangulation: Base edge $(i, j)$ forms a triangle with apex $k$, leaving sub-polygons $(i..k)$ and $(k..j)$: $\text{cost}(i, k, j) = v[i] \cdot v[k] \cdot v[j]$.
  • 1547. Minimum Cost to Cut a Stick: Adding sentinels $0$ and $n$ to `cuts`, the first cut at $c[k]$ inside stick segment $[c[i], c[j]]$ costs the segment's current length: $\text{cost}(i, k, j) = c[j] - c[i]$.
  • 312. Burst Balloons: Adding sentinel $1$s at both ends, if balloon $k$ is the last balloon burst inside $(i, j)$, its immediate surviving neighbors at the moment it bursts are guaranteed to be $i$ and $j$: $\text{cost}(i, k, j) = a[i] \cdot a[k] \cdot a[j]$.

1039. Minimum Score Triangulation of Polygon Medium · Canonical Open-Interval DP

State: dp[i][j] = min triangulation score of sub-polygon $i..j$ Time: $O(n^3)$ Space: $O(n^2)$

Recurrence & Intuition: Fix the base edge $(i, j)$ of the sub-polygon with vertices $i, i+1, \dots, j$. In any valid triangulation, edge $(i, j)$ must belong to exactly one triangle $(i, k, j)$ for some interior vertex $i < k < j$. That triangle contributes $v[i] \cdot v[k] \cdot v[j]$ and splits the remaining polygon into independent sub-polygons $i..k$ and $k..j$. When $j - i < 2$, fewer than $3$ vertices exist so `dp[i][j] = 0`. 

class Solution {
public:
    int minScoreTriangulation(vector<int>& v) {
        int n = v.size();
        vector<vector<int>> dp(n, vector<int>(n, 0));

        for (int len = 2; len < n; ++len) {
            for (int i = 0; i + len < n; ++i) {
                int j = i + len;
                dp[i][j] = INT_MAX;
                for (int k = i + 1; k < j; ++k) {
                    dp[i][j] = min(dp[i][j], dp[i][k] + dp[k][j] + v[i] * v[k] * v[j]);
                }
            }
        }
        return dp[0][n - 1];
    }
};

1547. Minimum Cost to Cut a Stick Hard · Interval DP Over Sorted Cut Coordinates

State: dp[i][j] = min cost to perform all cuts strictly between $c[i]$ and $c[j]$ Time: $O(m^3)$ where $m = |cuts| + 2$ Space: $O(m^2)$

Recurrence & Intuition: The stick length $n$ can be up to $10^6$, so we cannot index our DP table by raw stick coordinates. Instead, we only care about the at most $100$ cut positions! Insert sentinel endpoints $0$ and $n$ into `cuts` and sort it so $c[0] = 0 < c[1] < \dots < c[m-1] = n$. For interval $(i, j)$, if $c[k]$ ($i < k < j$) is chosen as the first cut inside $[c[i], c[j]]$, we pay $c[j] - c[i]$ plus `dp[i][k] + dp[k][j]`. 

class Solution {
public:
    int minCost(int n, vector<int>& cuts) {
        cuts.push_back(0);
        cuts.push_back(n);
        sort(cuts.begin(), cuts.end());

        int m = cuts.size();
        vector<vector<int>> dp(m, vector<int>(m, 0));

        for (int len = 2; len < m; ++len) {
            for (int i = 0; i + len < m; ++i) {
                int j = i + len;
                int best = INT_MAX;
                for (int k = i + 1; k < j; ++k) {
                    best = min(best, dp[i][k] + dp[k][j] + (cuts[j] - cuts[i]));
                }
                dp[i][j] = best;
            }
        }
        return dp[0][m - 1];
    }
};

312. Burst Balloons Hard · Reverse-Time Interval DP (Choose LAST Balloon)

State: dp[i][j] = max coins from bursting all balloons strictly inside $(i, j)$ Time: $O(n^3)$ Space: $O(n^2)$

Recurrence & Intuition: Why can't we pick which balloon $k$ to burst first? Because bursting $k$ first makes balloons $k-1$ and $k+1$ adjacent, coupling the left and right subproblems so they are no longer independent! Instead, pad the array with sentinel $1$s at indices $0$ and $m-1$, and ask: among all balloons strictly inside $(i, j)$, which balloon $k \in (i+1, j-1)$ is burst LAST? Since every other balloon in $(i, k)$ and $(k, j)$ has already been burst before $k$, at the moment $k$ bursts its immediate surviving neighbors are guaranteed to be the boundary sentinels $i$ and $j$, yielding $a[i] \cdot a[k] \cdot a[j]$ coins! 

class Solution {
public:
    int maxCoins(vector<int>& nums) {
        vector<int> a = {1};
        for (int x : nums) {
            if (x > 0) a.push_back(x); // zero-value balloons never contribute
        }
        a.push_back(1);

        int m = a.size();
        vector<vector<int>> dp(m, vector<int>(m, 0));

        for (int len = 2; len < m; ++len) {
            for (int i = 0; i + len < m; ++i) {
                int j = i + len;
                int best = 0;
                for (int k = i + 1; k < j; ++k) {
                    best = max(best, dp[i][k] + dp[k][j] + a[i] * a[k] * a[j]);
                }
                dp[i][j] = best;
            }
        }
        return dp[0][m - 1];
    }
};

2. Pattern 16: Partition DP — Divide the Input into Groups

In Partition DP, we must partition the array $A[0..n-1]$ into exactly $K$ contiguous non-empty groups. Unlike Interval DP (which splits $(i, j)$ into two arbitrary intervals), Partition DP always peels off the $k$-th group $[j..i-1]$ from the right end of prefix $0..i$, reducing to $k-1$ groups on prefix $0..j$: 

// Core Recurrence Shape:
// dp[k][i] = best cost to partition prefix A[0..i-1] into k groups
for (int k = 2; k <= K; ++k) {
    for (int i = k; i <= n; ++i) {
        for (int j = k - 1; j < i; ++j) {
            dp[k][i] = best(dp[k][i], combine(dp[k - 1][j], cost(j, i)));
        }
    }
}

410. Split Array Largest Sum Hard · Min-Max Partition DP

State: dp[g][i] = min largest subarray sum splitting $0..i-1$ into $g$ groups Time: $O(k \cdot n^2)$ DP (or $O(n \log(\sum))$ Binary Search) Space: $O(n)$ rolling row

Recurrence & Intuition: Let $\text{pref}[i]$ be the prefix sum of the first $i$ elements so $\text{cost}(j, i) = \text{pref}[i] - \text{pref}[j]$. To form $g$ groups out of the first $i$ elements, the last group spans $j..i-1$ for some $g-1 \le j < i$, and the largest sum among the $g$ groups is $\max(\text{dp}[g-1][j], \text{pref}[i] - \text{pref}[j])$. We minimize this over all valid $j$. 

class Solution {
public:
    int splitArray(vector<int>& nums, int k) {
        int n = nums.size();
        vector<long long> pref(n + 1, 0);
        for (int i = 0; i < n; ++i) {
            pref[i + 1] = pref[i] + nums[i];
        }

        // Base case g = 1: single group covering 0..i-1
        vector<long long> dp(n + 1, 0);
        for (int i = 1; i <= n; ++i) {
            dp[i] = pref[i];
        }

        for (int g = 2; g <= k; ++g) {
            vector<long long> nextDp(n + 1, LLONG_MAX);
            for (int i = g; i <= n; ++i) {
                for (int j = g - 1; j < i; ++j) {
                    long long candidate = max(dp[j], pref[i] - pref[j]);
                    nextDp[i] = min(nextDp[i], candidate);
                }
            }
            dp = move(nextDp);
        }
        return static_cast<int>(dp[n]);
    }
};

1335. Minimum Difficulty of a Job Schedule Hard · Range-Max Cost Partition DP

State: dp[day][i] = min difficulty scheduling first $i$ jobs in $\text{day}$ days Time: $O(d \cdot n^2)$ Space: $O(n)$

Recurrence & Intuition: Here $\text{cost}(j, i) = \max_{j \le p < i} \text{jobDifficulty}[p]$ (the hardest job done on day $g$). If we iterate the split point $j$ backward from $i-1$ down to $g-1$, we can maintain `curMax = max(curMax, jobDifficulty[j])` on the fly in $O(1)$ per step! 

class Solution {
public:
    int minDifficulty(vector<int>& jobDifficulty, int d) {
        int n = jobDifficulty.size();
        if (n < d) return -1;

        const int INF = 1e9;
        vector<int> dp(n + 1, INF);
        dp[0] = 0;

        for (int day = 1; day <= d; ++day) {
            vector<int> nextDp(n + 1, INF);
            for (int i = day; i <= n; ++i) {
                int curMax = 0;
                for (int j = i - 1; j >= day - 1; --j) {
                    curMax = max(curMax, jobDifficulty[j]);
                    if (dp[j] < INF) {
                        nextDp[i] = min(nextDp[i], dp[j] + curMax);
                    }
                }
            }
            dp = move(nextDp);
        }
        return dp[n];
    }
};
