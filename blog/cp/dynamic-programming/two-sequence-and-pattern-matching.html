    Two-Sequence Alignment & Pattern-Matching DP

    October 2026 · 26 min read · Patterns 7–8 (8 Problems)

    Whenever two strings or sequences $s$ (length $m$) and $t$ (length $n$) must be aligned, transformed, interleaved, or matched against a pattern, the canonical state space is a 2D table of prefix lengths $\text{dp}[i][j]$ over $s[0..i-1]$ and $t[0..j-1]$. Recognizing the $(m+1)\times(n+1)$ prefix grid is step one; choosing the exact transition for your objective is where the real work lies.

    Pattern 7: Two-Sequence DP — LCS and Alignment

    How to Recognize It

    Two strings or arrays must be compared element by element, and the optimal result for the full inputs depends on prefixes of both sequences.

    Core LCS Template

    if (s[i - 1] == t[j - 1])
    dp[i][j] = dp[i - 1][j - 1] + 1;
else
    dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);

    Connection #3: Same State Shape, Four Different Transitions

    Problem Objective When $s[i-1] == t[j-1]$ When $s[i-1] \neq t[j-1]$
    1143. LCS / 1035. Uncrossed Lines Maximize matches $\text{dp}[i-1][j-1] + 1$ $\max(\text{dp}[i-1][j],\; \text{dp}[i][j-1])$
    583. Delete Operation Minimize deletions $\text{dp}[i-1][j-1]$ $1 + \min(\text{dp}[i-1][j],\; \text{dp}[i][j-1])$
    72. Edit Distance Minimize ins/del/rep $\text{dp}[i-1][j-1]$ $1 + \min\{\text{dp}[i-1][j], \text{dp}[i][j-1], \text{dp}[i-1][j-1]\}$
    115. Distinct Subsequences Count subsequence ways $\text{dp}[i-1][j] + \text{dp}[i-1][j-1]$ $\text{dp}[i-1][j]$ (skip $s[i-1]$ only)

    ▶ Step-Through: 2D Prefix Table for LCS vs. Edit Distance

    Strings s = "abc" and t = "adc". Step through the 2D table cells $(i, j)$ to compare how LCS (maximize matches) and Edit Distance (minimize edits) fill the exact same grid.

    7.1 1143. Longest Common Subsequence

    State: $\text{dp}[i][j]$ = LCS length of $s[0..i-1]$ and $t[0..j-1]$. Using a 1D rolling array reduces space to $O(\min(m, n))$.

    class Solution {
public:
    int longestCommonSubsequence(string s, string t) {
        int m = s.size(), n = t.size();
        vector<int> dp(n + 1, 0);
        for (int i = 1; i <= m; ++i) {
            int prevDiag = 0;
            for (int j = 1; j <= n; ++j) {
                int temp = dp[j];
                if (s[i - 1] == t[j - 1]) dp[j] = prevDiag + 1;
                else dp[j] = max(dp[j], dp[j - 1]);
                prevDiag = temp;
            }
        }
        return dp[n];
    }
}; // Time: O(m * n), Space: O(n)

    7.2 1035. Uncrossed Lines

    Isomorphism to LCS: Drawing non-intersecting lines between equal numbers in `nums1` and `nums2` is 100% identical to finding the Longest Common Subsequence of the two integer arrays.

    class Solution {
public:
    int maxUncrossedLines(vector<int>& nums1, vector<int>& nums2) {
        int m = nums1.size(), n = nums2.size();
        vector<int> dp(n + 1, 0);
        for (int i = 1; i <= m; ++i) {
            int diag = 0;
            for (int j = 1; j <= n; ++j) {
                int tmp = dp[j];
                if (nums1[i - 1] == nums2[j - 1]) dp[j] = diag + 1;
                else dp[j] = max(dp[j], dp[j - 1]);
                diag = tmp;
            }
        }
        return dp[n];
    }
}; // Time: O(m * n), Space: O(n)

    7.3 583. Delete Operation for Two Strings

    Reduction: Making two strings equal using only deletions leaves their Longest Common Subsequence intact, so minimum deletions $= m + n - 2 \cdot \text{LCS}(s, t)$, or directly via edit-deletion DP:

    class Solution {
public:
    int minDistance(string word1, string word2) {
        int m = word1.size(), n = word2.size();
        vector<int> dp(n + 1);
        iota(dp.begin(), dp.end(), 0); // dp[0][j] = j deletions
        for (int i = 1; i <= m; ++i) {
            int diag = dp[0];
            dp[0] = i;
            for (int j = 1; j <= n; ++j) {
                int tmp = dp[j];
                if (word1[i - 1] == word2[j - 1]) dp[j] = diag;
                else dp[j] = 1 + min(dp[j], dp[j - 1]);
                diag = tmp;
            }
        }
        return dp[n];
    }
}; // Time: O(m * n), Space: O(n)

    7.4 72. Edit Distance

    Levenshtein Recurrence: When $w_1[i-1] \neq w_2[j-1]$, choose the minimum of deleting $w_1[i-1]$ ($\text{dp}[i-1][j] + 1$), inserting $w_2[j-1]$ ($\text{dp}[i][j-1] + 1$), or replacing $w_1[i-1] \to w_2[j-1]$ ($\text{dp}[i-1][j-1] + 1$).

    class Solution {
public:
    int minDistance(string word1, string word2) {
        int m = word1.size(), n = word2.size();
        vector<int> dp(n + 1);
        iota(dp.begin(), dp.end(), 0);
        for (int i = 1; i <= m; ++i) {
            int diag = dp[0];
            dp[0] = i;
            for (int j = 1; j <= n; ++j) {
                int tmp = dp[j];
                if (word1[i - 1] == word2[j - 1]) dp[j] = diag;
                else dp[j] = 1 + min({dp[j], dp[j - 1], diag});
                diag = tmp;
            }
        }
        return dp[n];
    }
}; // Time: O(m * n), Space: O(n)

    7.5 97. Interleaving String

    Feasibility State: $\text{dp}[i][j]$ = whether $s_1[0..i-1]$ and $s_2[0..j-1]$ can interleave to form $s_3[0..i+j-1]$.

    $$\text{dp}[i][j] = \big(\text{dp}[i-1][j] \land s_1[i-1] == s_3[i+j-1]\big) \;\lor\; \big(\text{dp}[i][j-1] \land s_2[j-1] == s_3[i+j-1]\big)$$ 
    class Solution {
public:
    bool isInterleave(string s1, string s2, string s3) {
        int m = s1.size(), n = s2.size();
        if (m + n != (int)s3.size()) return false;
        vector<char> dp(n + 1, 0);
        dp[0] = 1;
        for (int j = 1; j <= n; ++j)
            dp[j] = dp[j - 1] && (s2[j - 1] == s3[j - 1]);
        for (int i = 1; i <= m; ++i) {
            dp[0] = dp[0] && (s1[i - 1] == s3[i - 1]);
            for (int j = 1; j <= n; ++j) {
                dp[j] = (dp[j] && s1[i - 1] == s3[i + j - 1]) ||
                        (dp[j - 1] && s2[j - 1] == s3[i + j - 1]);
            }
        }
        return dp[n];
    }
}; // Time: O(m * n), Space: O(n)

    7.6 115. Distinct Subsequences

    Counting Subsequence Ways: $\text{dp}[i][j]$ = ways $t[0..j-1]$ appears as a subsequence of $s[0..i-1]$. Notice we can compressed to 1D iterating $j$ backward (just like 0/1 Knapsack!):

    class Solution {
public:
    int numDistinct(string s, string t) {
        int n = t.size();
        vector<unsigned long long> dp(n + 1, 0);
        dp[0] = 1; // empty target t matched in 1 way
        for (char c : s) {
            for (int j = n; j >= 1; --j) {
                if (c == t[j - 1]) dp[j] += dp[j - 1];
            }
        }
        return dp[n];
    }
}; // Time: O(m * n), Space: O(n)

    Pattern 8: Pattern-Matching DP — Wildcards and Regular Expressions

    How to Recognize It

    You must determine whether an input string $s$ matches a pattern $p$ containing special meta-characters. The state is $\text{dp}[i][j]$ = whether the first $i$ characters of $s$ match the first $j$ characters of $p$.

    Crucial Distinction (`*` Semantics):
    • Wildcard Matching (LC 44): * is a standalone token matching any sequence of characters (including empty). Transition at $p[j-1] == \text{'*'}$: $\text{dp}[i][j] = \text{dp}[i][j-1] \text{ (empty)} \lor \text{dp}[i-1][j] \text{ (consume } s[i-1]\text{)}$.
    • Regular Expression Matching (LC 10): * is a quantifier bound to the preceding pattern token $p[j-2]$, meaning zero or more repetitions of $p[j-2]$. Transition at $p[j-1] == \text{'*'}$: $\text{dp}[i][j] = \text{dp}[i][j-2] \text{ (zero repeats)} \lor (\text{match}(s[i-1], p[j-2]) \land \text{dp}[i-1][j])$.

    8.1 44. Wildcard Matching

    class Solution {
public:
    bool isMatch(string s, string p) {
        int m = s.size(), n = p.size();
        vector<vector<char>> dp(m + 1, vector<char>(n + 1, 0));
        dp[0][0] = 1;
        for (int j = 1; j <= n && p[j - 1] == '*'; ++j)
            dp[0][j] = 1; // leading '*' can match empty string
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (p[j - 1] == '*') {
                    dp[i][j] = dp[i][j - 1] || dp[i - 1][j];
                } else if (p[j - 1] == '?' || s[i - 1] == p[j - 1]) {
                    dp[i][j] = dp[i - 1][j - 1];
                }
            }
        }
        return dp[m][n];
    }
}; // Time: O(m * n), Space: O(m * n)

    8.2 10. Regular Expression Matching

    class Solution {
public:
    bool isMatch(string s, string p) {
        int m = s.size(), n = p.size();
        vector<vector<char>> dp(m + 1, vector<char>(n + 1, 0));
        dp[0][0] = 1;
        for (int j = 2; j <= n; ++j) {
            if (p[j - 1] == '*') dp[0][j] = dp[0][j - 2];
        }
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= n; ++j) {
                if (p[j - 1] == '*') {
                    dp[i][j] = dp[i][j - 2]; // zero occurrences of p[j-2]
                    if (p[j - 2] == '.' || p[j - 2] == s[i - 1])
                        dp[i][j] = dp[i][j] || dp[i - 1][j]; // consume one more char
                } else if (p[j - 1] == '.' || p[j - 1] == s[i - 1]) {
                    dp[i][j] = dp[i - 1][j - 1];
                }
            }
        }
        return dp[m][n];
    }
}; // Time: O(m * n), Space: O(m * n)
